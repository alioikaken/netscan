Стандартты socket және ThreadPoolExecutor кітапханалары негізінде жазылған консольдік утилита.

import socket
from datetime import datetime
from concurrent.futures import ThreadPoolExecutor, as_completed

PORTS = {
    21: ("FTP", "Қауіпті: деректер шифрланбай беріледі, анонимді кіруді жабу керек."),
    22: ("SSH", "Орташа қауіп: құпиясөздің орнына SSH-кілттерді қолданған дұрыс."),
    23: ("Telnet", "Өте қауіпті: логин мен құпиясөз ашық көрінеді, өшірген жөн."),
    25: ("SMTP", "Орташа қауіп: пошта серверін open relay-ге тексеру керек."),
    53: ("DNS", "Қалыпты: домендік атаулар қызметі."),
    80: ("HTTP", "Орташа қауіп: шифрланбаған трафик, 443 (HTTPS) қолданған жөн."),
    110: ("POP3", "Қауіпті: пошта трафигі шифрланбаған."),
    139: ("NetBIOS", "Қауіпті: сыртқы желіге ашық тұрмауы керек."),
    143: ("IMAP", "Қауіпті: пошта трафигі шифрланбаған."),
    443: ("HTTPS", "Қауіпсіз: шифрланған веб-трафик."),
    445: ("SMB", "Өте қауіпті: желілік осалдықтар жиі кездеседі, сыртқа жабу қажет."),
    3306: ("MySQL", "Қауіпті: ДҚБЖ сыртқа ашық болмауы тиіс, тек localhost-қа шектеу керек."),
    3389: ("RDP", "Өте қауіпті: қашықтан кіру порты, брутфорсқа жиі ұшырайды."),
    5432: ("PostgreSQL", "Қауіпті: деректер қоры сыртқа ашық тұр."),
    8080: ("HTTP-Alt", "Орташа қауіп: қосымша веб немесе админ-панель порты болуы мүмкін."),
}


def check_port(host_ip, port, timeout=1.0):
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.settimeout(timeout)
    try:
        if s.connect_ex((host_ip, port)) == 0:
            return port
    except OSError:
        return None
    finally:
        s.close()
    return None


def scan_target(target):
    try:
        ip = socket.gethostbyname(target)
    except socket.gaierror:
        print(f"Қате: '{target}' мекенжайын анықтау мүмкін болмады.")
        return

    print("=" * 60)
    print(f"Нысана: {target} ({ip})")
    print(f"Басталған уақыты: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("=" * 60)

    opened = []

    with ThreadPoolExecutor(max_workers=20) as pool:
        tasks = [pool.submit(check_port, ip, p) for p in PORTS]
        for task in as_completed(tasks):
            res = task.result()
            if res is not None:
                opened.append(res)

    opened.sort()

    if not opened:
        print("\nАшық порттар табылмады.")
    else:
        print(f"\nАшық порттар ({len(opened)}):")
        print("-" * 60)
        for p in opened:
            service, note = PORTS[p]
            print(f"Порт {p:<5} | {service:<10} | {note}")

    print("=" * 60)
    print("Тексеру аяқталды.")


if __name__ == "__main__":
    user_target = input("IP немесе домен енгізіңіз (мысалы, 127.0.0.1): ").strip()
    if not user_target:
        print("Мекенжай бос болмауы керек.")
    else:
        scan_target(user_target)
